## 🛡️ **Read vs. Write Access Control**

### **Updated Subscription Checker:**
```javascript
// middleware/subscriptionAuth.js
const checkSubscriptionAccess = (requiredPlan, feature, accessType = 'read') => {
  return async (req, res, next) => {
    try {
      const user = await User.findById(req.user.id);
      
      // Always allow READ access to existing data
      if (accessType === 'read') {
        req.user.subscriptionPlan = user.subscription_plan;
        req.user.subscriptionStatus = user.subscription_status;
        return next();
      }
      
      // Block WRITE/CREATE access if subscription inactive
      if (user.subscription_status !== 'active') {
        return res.status(403).json({
          error: 'Subscription required for new data',
          message: 'Please renew your subscription to add new trades, accounts, or journal entries.',
          currentData: 'Your existing data is safe and accessible',
          redirectTo: '/billing',
          accessType: 'read_only'
        });
      }
      
      // Check if subscription end date has passed
      if (new Date() > new Date(user.subscription_end_date)) {
        // Auto-expire the subscription but don't delete data
        await User.findByIdAndUpdate(req.user.id, {
          subscription_status: 'expired'
        });
        
        return res.status(403).json({
          error: 'Subscription expired - Read only access',
          message: 'Your subscription has expired. You can view existing data but cannot add new entries.',
          currentData: 'All your historical data remains accessible',
          redirectTo: '/billing',
          accessType: 'read_only'
        });
      }
      
      // Check plan access levels for new features
      const planLevels = {
        'starter': 1,
        'professional': 2,
        'elite': 3
      };
      
      const userLevel = planLevels[user.subscription_plan];
      const requiredLevel = planLevels[requiredPlan];
      
      if (userLevel < requiredLevel) {
        return res.status(403).json({
          error: 'Upgrade required for this feature',
          message: `This feature requires ${requiredPlan} plan or higher.`,
          currentPlan: user.subscription_plan,
          requiredPlan: requiredPlan,
          redirectTo: '/upgrade'
        });
      }
      
      req.user.subscriptionPlan = user.subscription_plan;
      req.user.subscriptionStatus = user.subscription_status;
      next();
    } catch (error) {
      res.status(500).json({ error: 'Subscription check failed' });
    }
  };
};
```

```javascript
// middleware/accountLimits.js
const checkAccountLimit = async (req, res, next) => {
  const user = await User.findById(req.user.id);
  const accountCount = await UserAccount.countDocuments({ 
    user_id: req.user.id, 
    is_active: true 
  });
  
  const limits = {
    'starter': 1,
    'professional': 5,
    'elite': Infinity
  };
  
  const currentLimit = limits[user.subscription_plan];
  
  if (accountCount >= currentLimit) {
    return res.status(403).json({
      error: 'Account limit reached',
      message: `Your ${user.subscription_plan} plan allows maximum ${currentLimit} accounts.`,
      currentCount: accountCount,
      maxAllowed: currentLimit,
      existingAccounts: 'Your existing accounts remain fully accessible',
      upgradeRequired: true,
      action: 'blocked_new_creation'
    });
  }
  
  next();
};
```

```javascript
// middleware/dataRetention.js
const enforceDataRetention = (req, res, next) => {
  const user = req.user;
  
  // If subscription is active, show data based on plan
  if (user.subscriptionStatus === 'active') {
    const retentionPeriods = {
      'starter': 30,    // 30 days
      'professional': 90,   // 90 days  
      'elite': Infinity     // Unlimited
    };
    
    const userPlan = user.subscriptionPlan;
    const retentionDays = retentionPeriods[userPlan];
    
    if (retentionDays !== Infinity) {
      const cutoffDate = new Date();
      cutoffDate.setDate(cutoffDate.getDate() - retentionDays);
      
      // Add date filter to query for active subscribers
      req.dataRetentionFilter = { 
        created_at: { $gte: cutoffDate } 
      };
    }
  } else {
    // If subscription expired, show limited recent data (last 7 days) as preview
    const previewDate = new Date();
    previewDate.setDate(previewDate.getDate() - 7);
    
    req.dataRetentionFilter = { 
      created_at: { $gte: previewDate } 
    };
    
    req.isPreviewMode = true;
  }
  
  next();
};
```
