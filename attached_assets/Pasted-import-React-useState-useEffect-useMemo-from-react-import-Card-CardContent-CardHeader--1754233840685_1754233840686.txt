                  <div className="bg-gradient-to-r from-blue-900/20 via-purple-900/10 to-blue-900/20 rounded-xl p-6 border border-blue-500/30">
                    <h4 className="text-white font-semibold text-lg mb-4 flex items-center gap-2">
                      <Settings className="w-5 h-5 text-blue-400" />
                      Trading Rules & Setup
                    </h4>
                    <div className="space-y-3">
                      {selectedStrategy.rules.split('\n').filter(rule => rule.trim()).map((rule, index) => (
                        <div key={index} className="flex items-start gap-3 p-3 bg-slate-800/40 rounded-lg">
                          <div className="flex items-center justify-center w-6 h-6 bg-blue-500 text-white rounded-full font-bold text-xs mt-0.5">
                            {index + 1}
                          </div>
                          <span className="text-slate-200 leading-relaxed flex-1">{rule.trim()}</span>
                        </div>
                      ))}
                    </div>
                  </div>
                )}

                {/* Context Information */}
                <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                  <div className="bg-slate-800/40 rounded-xl p-6 border border-slate-700/50">
                    <h4 className="text-white font-semibold mb-3 flex items-center gap-2">
                      <Activity className="w-5 h-5 text-green-400" />
                      Market Conditions
                    </h4>
                    <p className="text-slate-300">{selectedStrategy.marketConditions}</p>
                  </div>

                  <div className="bg-slate-800/40 rounded-xl p-6 border border-slate-700/50">
                    <h4 className="text-white font-semibold mb-3 flex items-center gap-2">
                      <DollarSign className="w-5 h-5 text-yellow-400" />
                      Target Assets
                    </h4>
                    <p className="text-slate-300">{selectedStrategy.assets}</p>
                  </div>
                </div>
              </div>
            )}

            <div className="flex justify-between items-center pt-6 mt-6 border-t border-slate-700">
              <div className="flex items-center gap-2 text-sm text-slate-500">
                <CheckCircle className="w-4 h-4" />
                <span>Strategy ready for deployment</span>
              </div>
              <div className="flex gap-3">
                <Button
                  variant="outline"
                  onClick={() => setIsDetailsDialogOpen(false)}
                  className="border-slate-600 text-slate-300 hover:bg-slate-700"
                >
                  Close
                </Button>
                <Button 
                  onClick={() => {
                    setSelectedStrategyId(selectedStrategy.id);
                    setIsDetailsDialogOpen(false);
                    setActiveTab('psychology');
                  }}
                  className="bg-gradient-to-r from-blue-500 to-purple-600 hover:from-blue-600 hover:to-purple-700"
                >
                  Use This Strategy
                </Button>
              </div>
            </div>
          </DialogContent>
        </Dialog>
      </div>
    </>
  );
}
