function shareOnTwitter() {
    const text = encodeURIComponent(document.getElementById('shareText').value);
    const url = encodeURIComponent(document.getElementById('shareUrl').value);
    window.open(`https://twitter.com/intent/tweet?text=${text}&url=${url}`, '_blank');
}

    <script>
        function shareOnTwitter() {
            const text = encodeURIComponent(document.getElementById('shareText').value);
            const url = encodeURIComponent(document.getElementById('shareUrl').value);
            window.open(`https://twitter.com/intent/tweet?text=${text}&url=${url}`, '_blank', 'width=600,height=400');
        }

        function shareOnFacebook() {
            const url = encodeURIComponent(document.getElementById('shareUrl').value);
            window.open(`https://www.facebook.com/sharer/sharer.php?u=${url}`, '_blank', 'width=600,height=400');
        }

        function openYouTube() {
            window.open('https://www.youtube.com', '_blank');
            alert('YouTube doesn\'t provide a direct share API for external content. Users can manually share your content once on the platform.');
        }

        function copyForTikTok() {
            const text = document.getElementById('shareText').value;
            const url = document.getElementById('shareUrl').value;
            const shareText = `${text}\n${url}`;
            
            navigator.clipboard.writeText(shareText).then(() => {
                alert('Content copied to clipboard! You can now paste it in the TikTok app.');
            }).catch(() => {
                prompt('Copy this text for TikTok:', shareText);
            });
        }

        function copyForInstagram() {
            const text = document.getElementById('shareText').value;
            const url = document.getElementById('shareUrl').value;
            const shareText = `${text}\n${url}`;
            
            navigator.clipboard.writeText(shareText).then(() => {
                alert('Content copied to clipboard! You can now paste it in the Instagram app.');
            }).catch(() => {
                prompt('Copy this text for Instagram:', shareText);
            });
        }
    </script>
