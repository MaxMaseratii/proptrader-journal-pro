### **Starter Plan Access Control:**
```javascript
// middleware/starterPlanLimits.js
const enforceStarterLimits = async (req, res, next) => {
  const user = req.user;
  
  if (user.subscriptionPlan !== 'starter') {
    return next();
  }
  
  const feature = req.params.feature || req.body.feature;
  
  switch (feature) {
    case 'dashboard':
      // Limit to 7 days history for starter
      const sevenDaysAgo = new Date();
      sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7);
      req.query.startDate = sevenDaysAgo.toISOString();
      break;
      
    case 'tradesLog':
      // Limit to last 30 trades
      req.query.limit = 30;
      req.query.sort = { created_at: -1 };
      break;
      
    case 'flowStateTraining':
      // Check monthly usage limit
      const usage = await FeatureUsage.findOne({
        user_id: user.id,
        feature_name: 'flow_state_training',
        month: new Date().getMonth(),
        year: new Date().getFullYear()
      });
      
      if (usage && usage.count >= 3) {
        return res.status(403).json({
          error: 'Monthly limit reached',
          message: 'Starter plan allows 3 Flow State Training sessions per month',
          currentUsage: usage.count,
          limit: 3,
          upgradeRequired: true
        });
      }
      break;
      
    case 'achievements':
    case 'chartsAnalytics':
    case 'tradingCompanion':
    case 'propFirmSpending':
    case 'payoutRecords':
    case 'reports':
      return res.status(403).json({
        error: 'Feature not available',
        message: `${feature} is not available in Starter plan`,
        currentPlan: 'starter',
        upgradeRequired: true,
        availableIn: ['professional', 'elite']
      });
  }
  
  next();
};
```

```javascript
// middleware/professionalPlanLimits.js
const enforceProfessionalLimits = async (req, res, next) => {
  const user = req.user;
  
  if (user.subscriptionPlan !== 'professional') {
    return next();
  }
  
  const feature = req.params.feature || req.body.feature;
  
  switch (feature) {
    case 'accounts':
      // Check 5 account limit
      const accountCount = await UserAccount.countDocuments({
        user_id: user.id,
        is_active: true
      });
      
      if (accountCount >= 5) {
        return res.status(403).json({
          error: 'Account limit reached',
          message: 'Professional plan allows maximum 5 accounts',
          currentCount: accountCount,
          limit: 5,
          upgradeToElite: true
        });
      }
      break;
      
    case 'dataRetention':
      // Limit to 90 days
      const ninetyDaysAgo = new Date();
      ninetyDaysAgo.setDate(ninetyDaysAgo.getDate() - 90);
      req.dataRetentionFilter = { created_at: { $gte: ninetyDaysAgo } };
      break;
      
    case 'chartsAnalytics':
      // Allow intermediate level only
      req.analyticsLevel = 'intermediate';
      break;
      
    case 'tradingCompanion':
      // Basic AI insights only
      req.aiLevel = 'basic';
      break;
      
    case 'reports':
      // Basic reports only
      req.reportLevel = 'basic';
      break;
      
    // ELITE-ONLY FEATURES
    case 'monteCarlo':
    case 'advancedReports':
    case 'unlimitedAccounts':
    case 'professionalAnalytics':
      return res.status(403).json({
        error: 'Elite feature',
        message: `${feature} is only available in Elite plan`,
        currentPlan: 'professional',
        upgradeRequired: true,
        availableIn: ['elite']
      });
  }
  
  next();
};
```

## 🔄 **Feature Usage Tracking**

### **Usage Counter for Limited Features:**
```javascript
// utils/featureUsage.js
const trackFeatureUsage = async (userId, featureName, limit = null) => {
  const currentMonth = new Date().getMonth();
  const currentYear = new Date().getFullYear();
  
  let usage = await FeatureUsage.findOne({
    user_id: userId,
    feature_name: featureName,
    month: currentMonth,
    year: currentYear
  });
  
  if (!usage) {
    usage = new FeatureUsage({
      user_id: userId,
      feature_name: featureName,
      month: currentMonth,
      year: currentYear,
      count: 0
    });
  }
  
  usage.count += 1;
  usage.last_used = new Date();
  await usage.save();
  
  if (limit && usage.count > limit) {
    throw new Error(`Monthly limit of ${limit} exceeded for ${featureName}`);
  }
  
  return usage;
};

// Reset monthly counters
const resetMonthlyUsage = async () => {
  const currentMonth = new Date().getMonth();
  const currentYear = new Date().getFullYear();
  
  await FeatureUsage.updateMany(
    {
      month: { $ne: currentMonth },
      year: { $ne: currentYear }
    },
    {
      count: 0,
      month: currentMonth,
      year: currentYear
    }
  );
};
```

```javascript
// components/PlanIndicators.jsx
const PlanIndicator = ({ user, feature }) => {
  const plan = user.subscriptionPlan;
  
  const getFeatureStatus = (feature, plan) => {
    const access = {
      starter: {
        achievements: false,
        chartsAnalytics: false,
        tradingCompanion: false,
        multipleAccounts: false,
        unlimitedData: false
      },
      professional: {
        achievements: true,
        chartsAnalytics: 'intermediate',
        tradingCompanion: 'basic',
        multipleAccounts: true,
        unlimitedData: false,
        monteCarlo: false,
        advancedReports: false
      },
      elite: {
        achievements: true,
        chartsAnalytics: 'professional',
        tradingCompanion: 'advanced',
        multipleAccounts: true,
        unlimitedData: true,
        monteCarlo: true,
        advancedReports: true
      }
    };
    
    return access[plan][feature];
  };
  
  const status = getFeatureStatus(feature, plan);
  
  if (status === false) {
    return (
      <div className="flex items-center gap-2 p-2 bg-amber-50 border border-amber-200 rounded">
        <Lock className="w-4 h-4 text-amber-600" />
        <span className="text-amber-700 text-sm">
          Upgrade to {plan === 'starter' ? 'Professional' : 'Elite'} to unlock
        </span>
      </div>
    );
  }
  
  if (typeof status === 'string') {
    return (
      <div className="flex items-center gap-2 p-2 bg-blue-50 border border-blue-200 rounded">
        <Info className="w-4 h-4 text-blue-600" />
        <span className="text-blue-700 text-sm">
          {status.charAt(0).toUpperCase() + status.slice(1)} level available
        </span>
      </div>
    );
  }
  
  return null;
};

// Account limit indicator
const AccountLimitIndicator = ({ user, currentCount }) => {
  const limits = {
    starter: 1,
    professional: 5,
    elite: Infinity
  };
  
  const limit = limits[user.subscriptionPlan];
  const percentage = limit === Infinity ? 0 : (currentCount / limit) * 100;
  
  return (
    <div className="p-3 bg-gray-50 rounded-lg">
      <div className="flex justify-between items-center mb-2">
        <span className="text-sm font-medium">Accounts Used</span>
        <span className="text-sm text-gray-600">
          {currentCount} / {limit === Infinity ? '∞' : limit}
        </span>
      </div>
      {limit !== Infinity && (
        <div className="w-full bg-gray-200 rounded-full h-2">
          <div 
            className={`h-2 rounded-full ${percentage >= 100 ? 'bg-red-400' : percentage >= 80 ? 'bg-amber-400' : 'bg-green-400'}`}
            style={{ width: `${Math.min(percentage, 100)}%` }}
          ></div>
        </div>
      )}
    </div>
  );
};
```
