```typescript
export const getValueColor = (value: number) => {
  if (value > 0) return colors.profit.primary;
  if (value < 0) return colors.loss.primary;
  return colors.neutral.primary;
};

export const formatCurrency = (value: number): string => {
  return new Intl.NumberFormat('en-US', {
    style: 'currency',
    currency: 'USD',
    minimumFractionDigits: 0,
    maximumFractionDigits: 0
  }).format(value);
};
```

**📄 `src/components/tradingview/TradingViewChart.tsx`**
```typescript
import React, { useEffect, useRef } from 'react';
import { createChart, IChartApi, ISeriesApi } from 'lightweight-charts';
import { Trade } from '../../types/trading';

interface TradingViewChartProps {
  trades: Trade[];
  symbol: string;
  height?: number;
  theme?: 'light' | 'dark';
}

export const TradingViewChart: React.FC<TradingViewChartProps> = ({
  trades,
  symbol,
  height = 400,
  theme = 'dark'
}) => {
  const chartContainerRef = useRef<HTMLDivElement>(null);
  const chartRef = useRef<IChartApi | null>(null);
  const seriesRef = useRef<ISeriesApi<'Candlestick'> | null>(null);

  useEffect(() => {
    if (!chartContainerRef.current) return;

    // Create chart
    const chart = createChart(chartContainerRef.current, {
      width: chartContainerRef.current.clientWidth,
      height: height,
      layout: {
        background: { color: theme === 'dark' ? '#1a1a1a' : '#ffffff' },
        textColor: theme === 'dark' ? '#d1d5db' : '#374151',
      },
      grid: {
        vertLines: { color: theme === 'dark' ? '#374151' : '#e5e7eb' },
        horzLines: { color: theme === 'dark' ? '#374151' : '#e5e7eb' },
      },
      timeScale: {
        timeVisible: true,
        secondsVisible: false,
      },
    });

    chartRef.current = chart;

    // Add candlestick series
    const candlestickSeries = chart.addCandlestickSeries({
      upColor: '#10b981',
      downColor: '#ef4444',
      borderDownColor: '#ef4444',
      borderUpColor: '#10b981',
      wickDownColor: '#ef4444',
      wickUpColor: '#10b981',
    });

    seriesRef.current = candlestickSeries;

    // Generate sample candlestick data (replace with real data)
    const sampleData = generateSampleCandlestickData();
    candlestickSeries.setData(sampleData);

    // Add trade markers
    addTradeMarkers(chart, trades);

    // Handle resize
    const handleResize = () => {
      if (chartContainerRef.current && chartRef.current) {
        chartRef.current.applyOptions({
          width: chartContainerRef.current.clientWidth,
        });
      }
    };

    window.addEventListener('resize', handleResize);

    return () => {
      window.removeEventListener('resize', handleResize);
      if (chartRef.current) {
        chartRef.current.remove();
      }
    };
  }, [trades, symbol, height, theme]);

  const addTradeMarkers = (chart: IChartApi, trades: Trade[]) => {
    const markers = trades.map(trade => ({
      time: new Date(trade.date).getTime() / 1000,
      position: trade.side === 'buy' ? 'belowBar' : 'aboveBar',
      color: trade.pnl > 0 ? '#10b981' : '#ef4444',
      shape: trade.side === 'buy' ? 'arrowUp' : 'arrowDown',
      text: `${trade.side.toUpperCase()} ${trade.symbol} @ ${trade.entryPrice} | P&L: ${trade.pnl}`,
    }));

    if (seriesRef.current) {
      seriesRef.current.setMarkers(markers as any);
    }
  };

  const generateSampleCandlestickData = () => {
    const data = [];
    let basePrice = 100;
    const startDate = new Date('2024-01-01');

    for (let i = 0; i < 100; i++) {
      const date = new Date(startDate);
      date.setDate(date.getDate() + i);
      
      const open = basePrice + (Math.random() - 0.5) * 2;
      const close = open + (Math.random() - 0.5) * 4;
      const high = Math.max(open, close) + Math.random() * 2;
      const low = Math.min(open, close) - Math.random() * 2;

      data.push({
        time: date.getTime() / 1000,
        open: parseFloat(open.toFixed(2)),
        high: parseFloat(high.toFixed(2)),
        low: parseFloat(low.toFixed(2)),
        close: parseFloat(close.toFixed(2)),
      });

      basePrice = close;
    }

    return data;
  };

  return (
    <div className="w-full bg-gray-900 rounded-lg p-4">
      <div className="flex items-center justify-between mb-4">
        <h3 className="text-lg font-semibold text-white">{symbol} Trading Chart</h3>
        <div className="flex items-center space-x-2">
          <span className="text-sm text-gray-400">Trades: {trades.length}</span>
        </div>
      </div>
      <div ref={chartContainerRef} className="w-full rounded-lg overflow-hidden" />
    </div>
  );
};
```

**📄 `src/components/tradingview/ChartGrid.tsx`**

```typescript
import React, { useState } from 'react';
import { TradingViewChart } from './TradingViewChart';
import { Trade } from '../../types/trading';

interface ChartGridProps {
  trades: Trade[];
  symbols: string[];
  gridSize?: 1 | 2 | 4;
}

export const ChartGrid: React.FC<ChartGridProps> = ({
  trades,
  symbols,
  gridSize = 2
}) => {
  const [selectedSymbol, setSelectedSymbol] = useState<string>(symbols[0] || '');

  const getTradesForSymbol = (symbol: string) => {
    return trades.filter(trade => trade.symbol === symbol);
  };

  const gridClasses = {
    1: 'grid-cols-1',
    2: 'grid-cols-1 lg:grid-cols-2',
    4: 'grid-cols-1 md:grid-cols-2 lg:grid-cols-4'
  };

  return (
    <div className="space-y-6">
      {/* Symbol Selector */}
      <div className="flex items-center space-x-4">
        <label className="text-gray-400 text-sm">Select Symbol:</label>
        <select
          value={selectedSymbol}
          onChange={(e) => setSelectedSymbol(e.target.value)}
          className="bg-gray-800 border border-gray-600 rounded-lg px-3 py-2 text-white"
        >
          {symbols.map(symbol => (
            <option key={symbol} value={symbol}>{symbol}</option>
          ))}
        </select>
      </div>

      {/* Single Chart View */}
      {gridSize === 1 && (
        <TradingViewChart
          trades={getTradesForSymbol(selectedSymbol)}
          symbol={selectedSymbol}
          height={500}
        />
      )}

      {/* Multi-Chart Grid */}
      {gridSize > 1 && (
        <div className={`grid ${gridClasses[gridSize]} gap-6`}>
          {symbols.slice(0, gridSize).map(symbol => (
            <TradingViewChart
              key={symbol}
              trades={getTradesForSymbol(symbol)}
              symbol={symbol}
              height={300}
            />
          ))}
        </div>
      )}
    </div>
  );
};
```

```typescript
    pnl: 947,

    status: 'closed'

export const useDashboardData = () => {
  const [filters, setFilters] = useState<DashboardFilters>({
    viewMode: 'all',
    selectedAccountIds: [],
    timePeriod: 'monthly',
    viewType: 'overview'
  });

  const [accounts] = useState<Account[]>(mockAccounts);
  const [trades] = useState<Trade[]>(mockTrades);

  const analytics = useMemo((): PortfolioMetrics => {
    let filteredAccounts = accounts;
    let filteredTrades = trades;

    // Apply filters
    if (filters.viewMode !== 'all' && filters.selectedAccountIds.length > 0) {
      filteredAccounts = accounts.filter(acc => 
        filters.selectedAccountIds.includes(acc.id)
      );
      filteredTrades = trades.filter(trade => 
        filters.selectedAccountIds.includes(trade.accountId)
      );
    }

    // Calculate metrics
    const totalBalance = filteredAccounts.reduce((sum, acc) => sum + acc.currentBalance, 0);
    const totalStarting = filteredAccounts.reduce((sum, acc) => sum + acc.startingBalance, 0);
    const totalPnl = totalBalance - totalStarting;
    
    const winningTrades = filteredTrades.filter(trade => trade.pnl > 0);
    const losingTrades = filteredTrades.filter(trade => trade.pnl < 0);
    const totalTrades = filteredTrades.length;
    
    const winRate = totalTrades > 0 ? (winningTrades.length / totalTrades) * 100 : 0;
    const averageWin = winningTrades.length > 0 ? 
      winningTrades.reduce((sum, trade) => sum + trade.pnl, 0) / winningTrades.length : 0;
    const averageLoss = losingTrades.length > 0 ? 
      Math.abs(losingTrades.reduce((sum, trade) => sum + trade.pnl, 0) / losingTrades.length) : 0;
    
    const profitFactor = averageLoss > 0 ? averageWin / averageLoss : 0;
    const rFactor = averageLoss > 0 ? averageWin / averageLoss : 0;

    return {
      totalBalance,
      totalPnl,
      winRate,
      totalTrades,
      winningTrades: winningTrades.length,
      losingTrades: losingTrades.length,
      bestTrade: Math.max(...filteredTrades.map(t => t.pnl), 0),
      worstTrade: Math.min(...filteredTrades.map(t => t.pnl), 0),
      averageWin,
      averageLoss,
      profitFactor,
      rFactor,
      disciplinedScore: 92 // Mock score
    };
  }, [accounts, trades, filters]);

  const actions = {
    setViewMode: (mode: DashboardFilters['viewMode']) => 
      setFilters(prev => ({ ...prev, viewMode: mode })),
    setSelectedAccountIds: (ids: number[]) => 
      setFilters(prev => ({ ...prev, selectedAccountIds: ids })),
    setTimePeriod: (period: DashboardFilters['timePeriod']) => 
      setFilters(prev => ({ ...prev, timePeriod: period })),
    setViewType: (type: DashboardFilters['viewType']) => 
      setFilters(prev => ({ ...prev, viewType: type }))
  };

  return {
    accounts,
    trades,
    analytics,
    filters,
    actions,
    isLoading: false,
    error: null
  };
};
```

```typescript
};
```

### **8.2 Error Boundaries**

**📄 `src/components/ui/ErrorBoundary.tsx`**
```typescript
import React, { Component, ErrorInfo, ReactNode } from 'react';

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
  error?: Error;
}

export class ErrorBoundary extends Component<Props, State> {
  public state: State = {
    hasError: false
  };

  public static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  public componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('Dashboard Error:', error, errorInfo);
  }

  public render() {
    if (this.state.hasError) {
      return (
        <div className="min-h-screen bg-gray-900 flex items-center justify-center">
          <div className="text-center text-white">
            <h2 className="text-2xl font-bold mb-4">Something went wrong</h2>
            <p className="text-gray-400 mb-4">
              {this.state.error?.message || 'An unexpected error occurred'}
            </p>
            <button
              onClick={() => window.location.reload()}
              className="bg-blue-600 hover:bg-blue-700 text-white px-4 py-2 rounded-lg"
            >
              Reload Dashboard
            </button>
          </div>
        </div>
      );
    }

    return this.props.children;
  }
}
```
