  // Enhanced auto-detect format with more patterns
  const detectFormat = (headers: string[]) => {
    const headerLower = headers.map(h => h.toLowerCase().trim());
    
    // Check for specific indicators
    if (headerLower.includes('orderid') || headerLower.some(h => h.includes('tradingview'))) return 'tradovate';
    if (headerLower.includes('ticket') || headerLower.includes('magic') || headerLower.includes('expert')) return 'mt4';
    if (headerLower.some(h => h.includes('rithmic') || h.includes('rith'))) return 'rithmic';
    if (headerLower.some(h => h.includes('cqg'))) return 'cqg';
    if (headerLower.includes('instrument') && headerLower.includes('action')) return 'ninjatrader';
    if (headerLower.includes('underlyingsymbol') || headerLower.includes('proceeds')) return 'interactive_brokers';
    if (headerLower.some(h => h.includes('ftmo') || h.includes('drawdown'))) return 'ftmo';
    if (headerLower.some(h => h.includes('topstep') || h.includes('combine'))) return 'topstep';
    if (headerLower.includes('exec time') || headerLower.includes('net amount')) return 'thinkorswim';
    if (headerLower.includes('executed qty') || headerLower.includes('commission asset')) return 'binance';
    if (headerLower.some(h => h.includes('myfunded') || h.includes('currency pair'))) return 'myfundedfx';
    if (headerLower.some(h => h.includes('project') || h.includes('custom'))) return 'project_x';
    
    return 'custom';
  };

  // Parse CSV data with enhanced error handling
  const parseCSV = (csvText: string) => {
    try {
      // Handle different line endings and separators
      const normalizedText = csvText.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
      const lines = normalizedText.trim().split('\n');
      
      // Auto-detect separator
      const firstLine = lines[0];
      const separators = [',', ';', '\t', '|'];
      const separator = separators.find(sep => firstLine.includes(sep)) || ',';
      
      const headers = firstLine.split(separator).map(h => h.trim().replace(/"/g, ''));
      const rows = lines.slice(1).map(line => {
        const values = line.split(separator).map(v => v.trim().replace(/"/g, ''));
        const row: Record<string, string> = {};
        headers.forEach((header, index) => {
          row[header] = values[index] || '';
        });
        return row;
      });

      return { headers, rows, separator };
    } catch (error) {
      console.error('CSV parsing error:', error);
      throw new Error('Invalid CSV format');
    }
  };

  // Enhanced trade grouping with better algorithms
  const groupOrdersIntoTrades = (orders: any[]) => {
    const trades = [];
    const processedOrders = new Set();
    
    // Sort orders by timestamp for better matching
    const sortedOrders = [...orders].sort((a, b) => 
      new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime()
    );
    
    sortedOrders.forEach((order, index) => {
      if (processedOrders.has(index)) return;
      
      const symbol = order.symbol;
      const side = order.side?.toLowerCase().trim();
      const quantity = Math.abs(parseFloat(order.quantity) || 0);
      const price = parseFloat(order.price) || 0;
      const timestamp = new Date(order.timestamp);
      
      if (!symbol || !side || !quantity || !price) return;
      
      // Look for opposing trade within reasonable time window
      const timeWindow = 48 * 60 * 60 * 1000; // 48 hours
      
      const matchingExit = sortedOrders.find((exitOrder, exitIndex) => {
        if (processedOrders.has(exitIndex) || exitIndex === index) return false;
        if (exitOrder.symbol !== symbol) return false;
        
        const exitSide = exitOrder.side?.toLowerCase().trim();
        const exitQty = Math.abs(parseFloat(exitOrder.quantity) || 0);
        const exitTime = new Date(exitOrder.timestamp);
        
        // Check if it's an opposing trade
        const isOpposingTrade = (side.includes('buy') && exitSide.includes('sell')) || 
                               (side.includes('sell') && exitSide.includes('buy'));
        
        // Check time window and quantity match
        const isWithinTimeWindow = Math.abs(exitTime.getTime() - timestamp.getTime()) <= timeWindow;
        const isSameQuantity = Math.abs(quantity - exitQty) < 0.01;
        
        return isOpposingTrade && isWithinTimeWindow && isSameQuantity;
      });

      if (matchingExit) {
        const entryPrice = price;
        const exitPrice = parseFloat(matchingExit.price) || 0;
        const multiplier = side.includes('buy') ? 1 : -1;
        const pnl = (exitPrice - entryPrice) * quantity * multiplier;
        
        trades.push({
          symbol,
          entryPrice,
          exitPrice,
          quantity,
          side: side.includes('buy') ? 'Long' : 'Short',
          pnl,
          entryTime: order.timestamp,
          exitTime: matchingExit.timestamp,
          entryOrder: order,
          exitOrder: matchingExit,
          duration: (new Date(matchingExit.timestamp).getTime() - timestamp.getTime()) / (1000 * 60 * 60) // hours
        });
        
        processedOrders.add(index);
        processedOrders.add(sortedOrders.indexOf(matchingExit));
      }
    });
    
    return trades;
  };

  // Enhanced behavioral analysis with advanced discipline metrics
  const analyzeTradingBehavior = (trades: any[], allOrders: any[]) => {
    if (!trades.length) return null;

    let totalPnL = 0;
    let winCount = 0;
    let lossCount = 0;
    const symbols = new Set();
    const dailyPnL: Record<string, number> = {};
    const weeklyPnL: Record<string, number> = {};
    const monthlyPnL: Record<string, number> = {};
    
    // Advanced discipline metrics
    let disciplinedTrades = 0;
    let modifiedTrades = 0;
    let breakEvenTrades = 0;
    let exceededTargets = 0;
    let exceededStops = 0;
    
    const tradeDetails = [];
    const timeAnalysis = {
      morning: 0,    // 6-12
      afternoon: 0,  // 12-18
      evening: 0,    // 18-24
      night: 0       // 0-6
    };

    trades.forEach(trade => {
      totalPnL += trade.pnl;
      symbols.add(trade.symbol);
      
      if (trade.pnl > 0) winCount++;
      else if (trade.pnl < 0) lossCount++;
      
      // Time analysis
      const hour = new Date(trade.entryTime).getHours();
      if (hour >= 6 && hour < 12) timeAnalysis.morning++;
      else if (hour >= 12 && hour < 18) timeAnalysis.afternoon++;
      else if (hour >= 18 && hour < 24) timeAnalysis.evening++;
      else timeAnalysis.night++;
      
      // P&L tracking
      const date = trade.exitTime.split(' ')[0];
      const weekKey = getWeekKey(new Date(date));
      const monthKey = getMonthKey(new Date(date));
      
      if (!dailyPnL[date]) dailyPnL[date] = 0;
      if (!weeklyPnL[weekKey]) weeklyPnL[weekKey] = 0;
      if (!monthlyPnL[monthKey]) monthlyPnL[monthKey] = 0;
      
      dailyPnL[date] += trade.pnl;
      weeklyPnL[weekKey] += trade.pnl;
      monthlyPnL[monthKey] += trade.pnl;
      
      // Check for discipline
      const entryTime = new Date(trade.entryTime);
      const exitTime = new Date(trade.exitTime);
      const cancelledInTrade = allOrders.filter(order => 
        order.status?.toLowerCase().includes('cancel') &&
        new Date(order.timestamp) >= entryTime &&
        new Date(order.timestamp) <= exitTime &&
        order.symbol === trade.symbol
      );
      
      const hadModifications = cancelledInTrade.length > 0;
      
      // Behavioral categorization
      let behaviorType = 'Unknown';
      if (hadModifications) {
        modifiedTrades++;
        
        // Check if moved to breakeven
        const priceDeviation = Math.abs(trade.exitPrice - trade.entryPrice) / trade.entryPrice;
        if (priceDeviation < 0.002) { // Less than 0.2% deviation
          breakEvenTrades++;
          behaviorType = 'Moved to Breakeven';
        } else {
          behaviorType = 'Modified SL/TP';
        }
      } else {
        disciplinedTrades++;
        behaviorType = 'Disciplined Exit';
      }
      
      // Check if exceeded targets or stops
      if (trade.pnl > 0) {
        const profitPercentage = Math.abs(trade.pnl) / (trade.entryPrice * trade.quantity);
        if (profitPercentage > 0.03) { // More than 3% profit
          exceededTargets++;
          behaviorType += ' (Exceeded Target)';
        }
      } else if (trade.pnl < 0) {
        const lossPercentage = Math.abs(trade.pnl) / (trade.entryPrice * trade.quantity);
        if (lossPercentage > 0.03) { // More than 3% loss
          exceededStops++;
          behaviorType += ' (Exceeded Stop)';
        }
      }
      
      tradeDetails.push({
        ...trade,
        hadModifications,
        cancelledOrders: cancelledInTrade.length,
        behaviorType
      });
    });

    const completedTrades = winCount + lossCount;
    const winRate = completedTrades > 0 ? ((winCount / completedTrades) * 100).toFixed(1) : 0;
    const disciplineScore = trades.length > 0 ? ((disciplinedTrades / trades.length) * 100).toFixed(1) : 0;
    const profitFactor = lossCount > 0 ? Math.abs(trades.filter(t => t.pnl > 0).reduce((sum, t) => sum + t.pnl, 0) / trades.filter(t => t.pnl < 0).reduce((sum, t) => sum + t.pnl, 0)).toFixed(2) : 'N/A';
    
    // Calculate streaks
    const streaks = calculateStreaks(trades);
    
    // Risk metrics
    const riskMetrics = calculateRiskMetrics(trades);

    return {
      totalTrades: trades.length,
      completedTrades,
      winCount,
      lossCount,
      winRate,
      disciplineScore,
      profitFactor,
      totalPnL: totalPnL.toFixed(2),
      avgPnL: completedTrades > 0 ? (totalPnL / completedTrades).toFixed(2) : 0,
      symbols: Array.from(symbols),
      dailyPnL,
      weeklyPnL,
      monthlyPnL,
      disciplinedTrades,
      modifiedTrades,
      breakEvenTrades,
      exceededTargets,
      exceededStops,
      tradeDetails: tradeDetails.slice(0, 15), // Show first 15 for preview
      bestTrade: trades.reduce((best, trade) => trade.pnl > best.pnl ? trade : best, trades[0]),
      worstTrade: trades.reduce((worst, trade) => trade.pnl < worst.pnl ? trade : worst, trades[0]),
      avgDuration: trades.reduce((sum, trade) => sum + trade.duration, 0) / trades.length,
      timeAnalysis,
      streaks,
      riskMetrics
    };
  };

  // Helper functions
  const getWeekKey = (date: Date) => {
    const year = date.getFullYear();
    const weekNum = Math.ceil(((date.getTime() - new Date(year, 0, 1).getTime()) / 86400000 + 1) / 7);
    return `${year}-W${weekNum}`;
  };

  const getMonthKey = (date: Date) => {
    return `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}`;
  };

  const calculateStreaks = (trades: any[]) => {
    let currentWinStreak = 0;
    let currentLossStreak = 0;
    let maxWinStreak = 0;
    let maxLossStreak = 0;
    
    trades.forEach(trade => {
      if (trade.pnl > 0) {
        currentWinStreak++;
        currentLossStreak = 0;
        maxWinStreak = Math.max(maxWinStreak, currentWinStreak);
      } else if (trade.pnl < 0) {
        currentLossStreak++;
        currentWinStreak = 0;
        maxLossStreak = Math.max(maxLossStreak, currentLossStreak);
      }
    });
    
    return { maxWinStreak, maxLossStreak, currentWinStreak, currentLossStreak };
  };

  const calculateRiskMetrics = (trades: any[]) => {
    const pnls = trades.map(t => t.pnl);
    const avgPnL = pnls.reduce((sum, pnl) => sum + pnl, 0) / pnls.length;
    
    // Standard deviation
    const variance = pnls.reduce((sum, pnl) => sum + Math.pow(pnl - avgPnL, 2), 0) / pnls.length;
    const stdDev = Math.sqrt(variance);
    
    // Sharpe ratio (assuming 0% risk-free rate)
    const sharpeRatio = stdDev > 0 ? (avgPnL / stdDev).toFixed(2) : 'N/A';
    
    // Maximum drawdown
    let maxDrawdown = 0;
    let peak = 0;
    let runningPnL = 0;
    
    trades.forEach(trade => {
      runningPnL += trade.pnl;
      if (runningPnL > peak) peak = runningPnL;
      const drawdown = peak - runningPnL;
      if (drawdown > maxDrawdown) maxDrawdown = drawdown;
    });
    
    return {
      sharpeRatio,
      maxDrawdown: maxDrawdown.toFixed(2),
      standardDeviation: stdDev.toFixed(2)
    };
  };

  // Apply column mapping with validation
  const applyMapping = (rows: any[], mapping: Record<string, string>) => {
    const mapped = rows.map(row => {
      const mappedRow: Record<string, any> = {};
      Object.keys(mapping).forEach(standardField => {
        const csvColumn = mapping[standardField];
        if (csvColumn && row[csvColumn] !== undefined) {
          mappedRow[standardField] = row[csvColumn];
        }
      });
      return mappedRow;
    }).filter(row => Object.keys(row).length > 0);
    
    setMappedData(mapped);
    
    // Analyze behavior if we have enough data
    if (mapped.length > 0) {
      const trades = groupOrdersIntoTrades(mapped);
      const analysis = analyzeTradingBehavior(trades, mapped);
      setBehaviorAnalysis(analysis);
    }
  };

  // Update column mapping
  const updateMapping = (standardField: string, csvColumn: string) => {
    const newMapping = { ...columnMapping, [standardField]: csvColumn };
    setColumnMapping(newMapping);
    if (parsedCsvData) {
      applyMapping(parsedCsvData.rows, newMapping);
    }
  };

  const handleFileSelect = (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    if (file && file.type === "text/csv") {
      setSelectedFile(file);
      setUploadResult(null);
      
      const reader = new FileReader();
      reader.onload = (e) => {
        const content = e.target?.result as string;
        
        try {
          // Parse and analyze CSV
          const parsed = parseCSV(content);
          setParsedCsvData(parsed);
          
          // Auto-detect format
          const format = detectFormat(parsed.headers);
          setDetectedFormat(format);
          
          // Auto-map if known format
          if (format !== 'custom' && formatTemplates[format as keyof typeof formatTemplates]) {
            const autoMapping: Record<string, string> = {};
            const template = formatTemplates[format as keyof typeof formatTemplates].fields;
            
            Object.keys(template).forEach(standardField => {
              const possibleColumns = template[standardField as keyof typeof template];
              const matchedColumn = parsed.headers.find(header => 
                possibleColumns.some(col => 
                  header.toLowerCase().includes(col.toLowerCase())
                )
              );
              if (matchedColumn) {
                autoMapping[standardField] = matchedColumn;
              }
            });
            
            setColumnMapping(autoMapping);
            applyMapping(parsed.rows, autoMapping);
          }
          
          toast({
            title: "CSV Parsed Successfully",
            description: `Detected ${format !== 'custom' ? formatTemplates[format as keyof typeof formatTemplates]?.name : 'custom'} format with ${parsed.rows.length} records.`,
          });
        } catch (error) {
          console.error('CSV processing error:', error);
          toast({
            title: "CSV Processing Error",
            description: "Failed to process CSV file. Please check the format.",
            variant: "destructive",
          });
        }
      };
      reader.readAsText(file);
    } else {
      toast({
        title: "Invalid File",
        description: "Please select a valid CSV file.",
        variant: "destructive",
      });
    }
  };

  const handleUpload = async () => {
    if (!selectedFile || !selectedAccount) {
      toast({
        title: "Missing Information",
        description: "Please select both a file and an account.",
        variant: "destructive",
      });
      return;
    }

    if (!mappedData.length) {
      toast({
        title: "No Data Mapped",
        description: "Please map the columns before importing.",
        variant: "destructive",
      });
      return;
    }

    setIsUploading(true);

    try {
      const fileContent = await selectedFile.text();
      
      const response = await fetch('/api/trades/import-csv', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          accountId: parseInt(selectedAccount),
          csvData: fileContent,
          columnMapping: Object.keys(columnMapping).length > 0 ? columnMapping : undefined,
          detectedFormat: detectedFormat || undefined,
          behaviorAnalysis: behaviorAnalysis || undefined,
          mappedRecords: mappedData.length
        }),
      });

      if (!response.ok) {
        throw new Error('Upload failed');
      }

      const result = await response.json();
      setUploadResult(result);
      
      toast({
        title: "Upload Complete",
        description: `Successfully imported ${result.recordsImported} trades out of ${result.recordsProcessed} records. ${behaviorAnalysis ? `Discipline Score: ${behaviorAnalysis.disciplineScore}%` : ''}`,
      });
    } catch (error) {
      console.error('Upload error:', error);
      toast({
        title: "Upload Failed",
        description: "There was an error importing your CSV file.",
        variant: "destructive",
      });
    } finally {
      setIsUploading(false);
    }
  };

  return (
    <div className="p-6 space-y-8 bg-prop-gradient-main min-h-screen">
      {/* Header */}
      <div className="flex items-center justify-between">
        <div className="flex items-center space-x-4">
          <Button 
            variant="outline" 
            size="sm"
            onClick={() => setLocation('/dashboard-showcase')}
            className="border-prop-gold/20 hover:bg-prop-gold/10"
          >
            <ArrowLeft className="h-4 w-4 mr-2" />
            Back to Dashboard
          </Button>
          <div>
            <h1 className="text-3xl font-bold text-gradient-rainbow">Universal CSV Import</h1>
            <p className="text-gray-400">Import your trading data from any broker with advanced behavioral analysis</p>
          </div>
        </div>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-2 gap-8">
        {/* Upload Section */}
        <div className="space-y-6">
          <Card className="bg-prop-card border-prop-tiffany/20">
            <CardHeader>
              <CardTitle className="text-gradient-rainbow flex items-center">
                <Upload className="mr-2 h-5 w-5" />
                Upload CSV File
              </CardTitle>
              <p className="text-gray-400 text-sm">Auto-detects 12+ broker formats with intelligent column mapping</p>
            </CardHeader>
            <CardContent className="space-y-6">
              {/* Account Selection */}
              <div>
                <Label className="text-sm font-medium text-gray-300 mb-2 block">
                  Select Trading Account
                </Label>
                <Select value={selectedAccount} onValueChange={setSelectedAccount}>
                  <SelectTrigger className="bg-prop-dark border-prop-tiffany/20">
                    <SelectValue placeholder="Choose an account..." />
                  </SelectTrigger>
                  <SelectContent>
                    {accounts.map((account) => (
                      <SelectItem key={account.id} value={account.id.toString()}>
                        {account.name} - {account.firm} ({account.type})
                      </SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </div>

              {/* File Upload */}
              <div>
                <Label className="text-sm font-medium text-gray-300 mb-2 block">
                  CSV File
                </Label>
                <div className="border-2 border-dashed border-prop-tiffany/20 rounded-lg p-6 text-center hover:border-prop-tiffany/40 transition-colors">
                  <input
                    type="file"
                    accept=".csv"
                    onChange={handleFileSelect}
                    className="hidden"
                    id="csv-upload"
                  />
                  <label htmlFor="csv-upload" className="cursor-pointer">
                    <FileText className="mx-auto h-12 w-12 text-prop-tiffany mb-4" />
                    <p className="text-gray-300 mb-2">
                      {selectedFile ? selectedFile.name : "Click to select CSV file"}
                    </p>
                    <p className="text-sm text-gray-400">
                      Universal format support with behavioral analysis
                    </p>
                  </label>
                </div>
                
                {selectedFile && detectedFormat && (
                  <div className="mt-4 p-4 bg-prop-gradient-subtle rounded-lg border border-prop-tiffany/20">
                    <div className="flex items-center justify-between mb-2">
                      <span className="text-gray-300 font-medium">Detected Format:</span>
                      <Badge className={`${formatTemplates[detectedFormat as keyof typeof formatTemplates]?.color || 'bg-gray-600'} text-white`}>
                        {formatTemplates[detectedFormat as keyof typeof formatTemplates]?.name || detectedFormat}
                      </Badge>
                    </div>
                    <p className="text-sm text-gray-400">
                      {formatTemplates[detectedFormat as keyof typeof formatTemplates]?.description || 'Custom format detected'}
                    </p>
                  </div>
                )}
              </div>

              {/* Upload Button */}
              <Button
                onClick={handleUpload}
                disabled={!selectedFile || !selectedAccount || isUploading || !mappedData.length}
                className="w-full bg-prop-gradient-tiffany text-black font-bold hover-scale"
              >
                {isUploading ? "Analyzing & Importing..." : "Import CSV Data"}
              </Button>
            </CardContent>
          </Card>

          {/* Column Mapping */}
          {parsedCsvData && (
            <Card className="bg-prop-card border-prop-green/20">
              <CardHeader>
                <CardTitle className="text-gradient-rainbow flex items-center justify-between">
                  <div className="flex items-center">
                    <Settings className="mr-2 h-5 w-5" />
                    Column Mapping
                  </div>
                  <Button
                    variant="outline"
                    size="sm"
                    onClick={() => setShowAdvanced(!showAdvanced)}
                    className="text-gray-300 border-gray-600"
                  >
                    {showAdvanced ? 'Hide Advanced' : 'Show Advanced'}
                  </Button>
                </CardTitle>
              </CardHeader>
              <CardContent>
                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  {['symbol', 'side', 'quantity', 'price', 'timestamp', 'status'].map(field => (
                    <div key={field} className="flex flex-col">
                      <Label className="text-sm font-medium text-gray-300 mb-1 capitalize">
                        {field} {field === 'symbol' || field === 'side' || field === 'quantity' || field === 'price' ? '*' : ''}
                      </Label>
                      <Select
                        value={columnMapping[field] || ''}
                        onValueChange={(value) => updateMapping(field, value)}
                      >
                        <SelectTrigger className="bg-prop-dark border-prop-tiffany/20">
                          <SelectValue placeholder="Select column..." />
                        </SelectTrigger>
                        <SelectContent>
                          {parsedCsvData.headers.map((header: string) => (
                            <SelectItem key={header} value={header}>
                              {header}
                            </SelectItem>
                          ))}
                        </SelectContent>
                      </Select>
                    </div>
                  ))}
                  
                  {showAdvanced && ['orderId', 'account', 'type'].map(field => (
                    <div key={field} className="flex flex-col">
                      <Label className="text-sm font-medium text-gray-300 mb-1 capitalize">
                        {field} (Optional)
                      </Label>
                      <Select
                        value={columnMapping[field] || ''}
                        onValueChange={(value) => updateMapping(field, value)}
                      >
                        <SelectTrigger className="bg-prop-dark border-prop-tiffany/20">
                          <SelectValue placeholder="Select column..." />
                        </SelectTrigger>
                        <SelectContent>
                          {parsedCsvData.headers.map((header: string) => (
                            <SelectItem key={header} value={header}>
                              {header}
                            </SelectItem>
                          ))}
                        </SelectContent>
                      </Select>
                    </div>
                  ))}
                </div>
              </CardContent>
            </Card>
          )}
        </div>

        {/* Results & Analysis Section */}
        <div className="space-y-6">
          {/* Import Results */}
          <Card className="bg-prop-card border-prop-green/20">
            <CardHeader>
              <CardTitle className="text-gradient-rainbow flex items-center">
                <CheckCircle className="mr-2 h-5 w-5" />
                Import Results
              </CardTitle>
            </CardHeader>
            <CardContent>
              {uploadResult ? (
                <div className="space-y-4">
                  <div className="flex items-center justify-between p-3 bg-prop-gradient-subtle rounded-lg">
                    <span className="text-gray-300">Records Processed</span>
                    <span className="text-prop-green font-bold">{uploadResult.recordsProcessed}</span>
                  </div>
                  <div className="flex items-center justify-between p-3 bg-prop-gradient-subtle rounded-lg">
                    <span className="text-gray-300">Records Imported</span>
                    <span className="text-prop-green font-bold">{uploadResult.recordsImported}</span>
                  </div>
                  <div className="flex items-center justify-between p-3 bg-prop-gradient-subtle rounded-lg">
                    <span className="text-gray-300">Success Rate</span>
                    <span className="text-prop-gold font-bold">
                      {Math.round((uploadResult.recordsImported / uploadResult.recordsProcessed) * 100)}%
                    </span>
                  </div>
                  
                  {uploadResult.errors && uploadResult.errors.length > 0 && (
                    <div className="mt-4">
                      <h4 className="text-prop-pink font-semibold mb-2 flex items-center">
                        <AlertCircle className="mr-2 h-4 w-4" />
                        Import Errors
                      </h4>
                      <div className="max-h-32 overflow-y-auto space-y-1">
                        {uploadResult.errors.map((error: string, index: number) => (
                          <p key={index} className="text-sm text-gray-400 p-2 bg-prop-dark rounded">
                            {error}
                          </p>
                        ))}
                      </div>
                    </div>
                  )}
                </div>
              ) : (
                <div className="text-center py-8">
                  <FileText className="mx-auto h-16 w-16 text-gray-500 mb-4" />
                  <p className="text-gray-400">Upload a CSV file to see import results</p>
                </div>
              )}
            </CardContent>
          </Card>

          {/* Advanced Behavioral Analysis */}
          {behaviorAnalysis && (
            <Card className="bg-prop-card border-prop-blue/20">
              <CardHeader>
                <CardTitle className="text-gradient-rainbow flex items-center justify-between">
                  <div className="flex items-center">
                    <Brain className="mr-2 h-5 w-5" />
                    Advanced Trading Analysis
                  </div>
                  <Button
                    variant="outline"
                    size="sm"
                    onClick={() => setShowDisciplineDetails(!showDisciplineDetails)}
                    className="text-gray-300 border-gray-600"
                  >
                    {showDisciplineDetails ? 'Hide Details' : 'Show Details'}
                  </Button>
                </CardTitle>
              </CardHeader>
              <CardContent>
                {/* Core Metrics */}
                <div className="grid grid-cols-2 md:grid-cols-4 gap-4 mb-6">
                  <div className="bg-prop-gradient-subtle rounded-lg p-4">
                    <div className="text-2xl font-bold text-prop-green">{behaviorAnalysis.totalTrades}</div>
                    <div className="text-sm text-gray-400">Total Trades</div>
                  </div>
                  <div className="bg-prop-gradient-subtle rounded-lg p-4">
                    <div className="text-2xl font-bold text-prop-tiffany">{behaviorAnalysis.winRate}%</div>
                    <div className="text-sm text-gray-400">Win Rate</div>
                  </div>
                  <div className="bg-prop-gradient-subtle rounded-lg p-4">
                    <div className={`text-2xl font-bold ${parseFloat(behaviorAnalysis.disciplineScore) >= 70 ? 'text-prop-green' : parseFloat(behaviorAnalysis.disciplineScore) >= 50 ? 'text-prop-gold' : 'text-prop-pink'}`}>
                      {behaviorAnalysis.disciplineScore}%
                    </div>
                    <div className="text-sm text-gray-400">Discipline Score</div>
                  </div>
                  <div className="bg-prop-gradient-subtle rounded-lg p-4">
                    <div className={`text-2xl font-bold ${parseFloat(behaviorAnalysis.totalPnL) >= 0 ? 'text-prop-green' : 'text-prop-pink'}`}>
                      ${behaviorAnalysis.totalPnL}
                    </div>
                    <div className="text-sm text-gray-400">Total P&L</div>
                  </div>
                </div>

                {/* Discipline Breakdown */}
                <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-6">
                  <div className="bg-prop-gradient-subtle rounded-lg p-4 border border-prop-green/20">
                    <div className="flex items-center justify-between mb-2">
                      <span className="text-prop-green font-medium">Disciplined Trades</span>
                      <span className="text-2xl font-bold text-prop-green">{behaviorAnalysis.disciplinedTrades}</span>
                    </div>
                    <div className="text-sm text-gray-400">Followed original plan</div>
                  </div>
                  <div className="bg-prop-gradient-subtle rounded-lg p-4 border border-prop-pink/20">
                    <div className="flex items-center justify-between mb-2">
                      <span className="text-prop-pink font-medium">Modified Trades</span>
                      <span className="text-2xl font-bold text-prop-pink">{behaviorAnalysis.modifiedTrades}</span>
                    </div>
                    <div className="text-sm text-gray-400">Changed SL/TP levels</div>
                  </div>
                </div>

                {/* Extended Analysis */}
                {showDisciplineDetails && (
                  <div className="space-y-6">
                    {/* Risk Metrics */}
                    <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
                      <div className="bg-prop-gradient-subtle rounded-lg p-4">
                        <div className="text-lg font-bold text-prop-blue">{behaviorAnalysis.profitFactor}</div>
                        <div className="text-sm text-gray-400">Profit Factor</div>
                      </div>
                      <div className="bg-prop-gradient-subtle rounded-lg p-4">
                        <div className="text-lg font-bold text-prop-purple">{behaviorAnalysis.riskMetrics?.sharpeRatio || 'N/A'}</div>
                        <div className="text-sm text-gray-400">Sharpe Ratio</div>
                      </div>
                      <div className="bg-prop-gradient-subtle rounded-lg p-4">
                        <div className="text-lg font-bold text-prop-orange">${behaviorAnalysis.riskMetrics?.maxDrawdown || '0.00'}</div>
                        <div className="text-sm text-gray-400">Max Drawdown</div>
                      </div>
                    </div>

                    {/* Trading Time Analysis */}
                    <div className="bg-prop-gradient-subtle rounded-lg p-4">
                      <h4 className="font-semibold text-white mb-3">Trading Time Distribution</h4>
                      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
                        <div className="text-center">
                          <div className="text-lg font-bold text-prop-yellow">{behaviorAnalysis.timeAnalysis.morning}</div>
                          <div className="text-sm text-gray-400">Morning (6-12)</div>
                        </div>
                        <div className="text-center">
                          <div className="text-lg font-bold text-prop-green">{behaviorAnalysis.timeAnalysis.afternoon}</div>
                          <div className="text-sm text-gray-400">Afternoon (12-18)</div>
                        </div>
                        <div className="text-center">
                          <div className="text-lg font-bold text-prop-blue">{behaviorAnalysis.timeAnalysis.evening}</div>
                          <div className="text-sm text-gray-400">Evening (18-24)</div>
                        </div>
                        <div className="text-center">
                          <div className="text-lg font-bold text-prop-purple">{behaviorAnalysis.timeAnalysis.night}</div>
                          <div className="text-sm text-gray-400">Night (0-6)</div>
                        </div>
                      </div>
                    </div>

                    {/* Streaks */}
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                      <div className="bg-prop-gradient-subtle rounded-lg p-4 border border-prop-green/20">
                        <div className="flex items-center justify-between mb-2">
                          <span className="text-prop-green font-medium">Max Win Streak</span>
                          <span className="text-2xl font-bold text-prop-green">{behaviorAnalysis.streaks?.maxWinStreak || 0}</span>
                        </div>
                        <div className="text-sm text-gray-400">Consecutive winning trades</div>
                      </div>
                      <div className="bg-prop-gradient-subtle rounded-lg p-4 border border-prop-pink/20">
                        <div className="flex items-center justify-between mb-2">
                          <span className="text-prop-pink font-medium">Max Loss Streak</span>
                          <span className="text-2xl font-bold text-prop-pink">{behaviorAnalysis.streaks?.maxLossStreak || 0}</span>
                        </div>
                        <div className="text-sm text-gray-400">Consecutive losing trades</div>
                      </div>
                    </div>

                    {/* Best/Worst Trades */}
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                      <div className="bg-prop-gradient-subtle rounded-lg p-4 border border-prop-green/20">
                        <div className="flex items-center mb-2">
                          <TrendingUp className="h-4 w-4 text-prop-green mr-2" />
                          <span className="text-prop-green font-medium">Best Trade</span>
                        </div>
                        <div className="text-sm text-gray-300">
                          {behaviorAnalysis.bestTrade.symbol} {behaviorAnalysis.bestTrade.side}: ${behaviorAnalysis.bestTrade.pnl.toFixed(2)}
                        </div>
                      </div>
                      <div className="bg-prop-gradient-subtle rounded-lg p-4 border border-prop-pink/20">
                        <div className="flex items-center mb-2">
                          <TrendingDown className="h-4 w-4 text-prop-pink mr-2" />
                          <span className="text-prop-pink font-medium">Worst Trade</span>
                        </div>
                        <div className="text-sm text-gray-300">
                          {behaviorAnalysis.worstTrade.symbol} {behaviorAnalysis.worstTrade.side}: ${behaviorAnalysis.worstTrade.pnl.toFixed(2)}
                        </div>
                      </div>
                    </div>

                    {/* Instruments Traded */}
                    <div className="bg-prop-gradient-subtle rounded-lg p-4">
                      <h4 className="font-semibold text-white mb-3">Instruments Traded ({behaviorAnalysis.symbols.length})</h4>
                      <div className="flex flex-wrap gap-2">
                        {behaviorAnalysis.symbols.map((symbol: string) => (
                          <Badge key={symbol} className="bg-prop-blue text-white">
                            {symbol}
                          </Badge>
                        ))}
                      </div>
                    </div>
                  </div>
                )}
              </CardContent>
            </Card>
          )}
        </div>
      </div>

      {/* Trade-by-Trade Analysis */}
      {behaviorAnalysis && behaviorAnalysis.tradeDetails && (
        <Card className="bg-prop-card border-prop-gold/20">
          <CardHeader>
            <CardTitle className="text-gradient-rainbow flex items-center justify-between">
              <div className="flex items-center">
                <Activity className="mr-2 h-5 w-5" />
                Trade-by-Trade Analysis
              </div>
              <Button
                variant="outline"
                size="sm"
                onClick={() => setShowTradeDetails(!showTradeDetails)}
                className="text-gray-300 border-gray-600"
              >
                {showTradeDetails ? 'Hide Details' : 'Show Details'}
              </Button>
            </CardTitle>
          </CardHeader>
          <CardContent>
            {showTradeDetails && (
              <div className="overflow-x-auto">
                <table className="min-w-full table-auto">
                  <thead className="bg-prop-gradient-subtle">
                    <tr>
                      <th className="px-4 py-2 text-left text-xs font-medium text-gray-300 uppercase">Symbol</th>
                      <th className="px-4 py-2 text-left text-xs font-medium text-gray-300 uppercase">Side</th>
                      <th className="px-4 py-2 text-left text-xs font-medium text-gray-300 uppercase">Entry</th>
                      <th className="px-4 py-2 text-left text-xs font-medium text-gray-300 uppercase">Exit</th>
                      <th className="px-4 py-2 text-left text-xs font-medium text-gray-300 uppercase">P&L</th>
                      <th className="px-4 py-2 text-left text-xs font-medium text-gray-300 uppercase">Duration</th>
                      <th className="px-4 py-2 text-left text-xs font-medium text-gray-300 uppercase">Behavior</th>
                      <th className="px-4 py-2 text-left text-xs font-medium text-gray-300 uppercase">Discipline</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-gray-700">
                    {behaviorAnalysis.tradeDetails.map((trade: any, index: number) => (
                      <tr key={index} className="hover:bg-prop-gradient-subtle">
                        <td className="px-4 py-2 text-sm font-medium text-white">{trade.symbol}</td>
                        <td className="px-4 py-2 text-sm">
                          <span className={`px-2 py-1 rounded text-xs font-medium ${
                            trade.side === 'Long' ? 'bg-prop-green text-white' : 'bg-prop-pink text-white'
                          }`}>
                            {trade.side}
                          </span>
                        </td>
                        <td className="px-4 py-2 text-sm text-gray-300">${trade.entryPrice.toFixed(2)}</td>
                        <td className="px-4 py-2 text-sm text-gray-300">${trade.exitPrice.toFixed(2)}</td>
                        <td className="px-4 py-2 text-sm">
                          <span className={`font-medium ${trade.pnl >= 0 ? 'text-prop-green' : 'text-prop-pink'}`}>
                            ${trade.pnl.toFixed(2)}
                          </span>
                        </td>
                        <td className="px-4 py-2 text-sm text-gray-300">{trade.duration.toFixed(1)}h</td>
                        <td className="px-4 py-2 text-sm">
                          <span className={`px-2 py-1 rounded text-xs font-medium ${
                            trade.behaviorType.includes('Disciplined') ? 'bg-prop-green text-white' :
                            trade.behaviorType.includes('Breakeven') ? 'bg-prop-blue text-white' :
                            trade.behaviorType.includes('Exceeded Target') ? 'bg-prop-tiffany text-black' :
                            trade.behaviorType.includes('Exceeded Stop') ? 'bg-prop-pink text-white' :
                            'bg-prop-gold text-black'
                          }`}>
                            {trade.behaviorType}
                          </span>
                        </td>
                        <td className="px-4 py-2 text-sm">
                          {trade.hadModifications ? (
                            <AlertCircle className="h-4 w-4 text-prop-orange" title="Modified SL/TP" />
                          ) : (
                            <CheckCircle className="h-4 w-4 text-prop-green" title="Original Plan Followed" />
                          )}
                        </td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            )}
          </CardContent>
        </Card>
      )}

      {/* Data Preview */}
      {mappedData.length > 0 && (
        <Card className="bg-prop-card border-prop-gold/20">
          <CardHeader>
            <CardTitle className="text-gradient-rainbow flex items-center">
              <Target className="mr-2 h-5 w-5" />
              Data Preview ({mappedData.length} records mapped)
            </CardTitle>
          </CardHeader>
          <CardContent>
            <div className="overflow-x-auto">
              <table className="min-w-full table-auto">
                <thead className="bg-prop-gradient-subtle">
                  <tr>
                    {Object.keys(columnMapping).filter(field => columnMapping[field]).map(field => (
                      <th key={field} className="px-4 py-2 text-left text-xs font-medium text-gray-300 uppercase tracking-wider">
                        {field}
                      </th>
                    ))}
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-700">
                  {mappedData.slice(0, 10).map((row, index) => (
                    <tr key={index} className="hover:bg-prop-gradient-subtle">
                      {Object.keys(columnMapping).filter(field => columnMapping[field]).map(field => (
                        <td key={field} className="px-4 py-2 whitespace-nowrap text-sm text-gray-300">
                          {row[field] || '-'}
                        </td>
                      ))}
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
            {mappedData.length > 10 && (
              <p className="text-sm text-gray-400 mt-4">
                Showing first 10 of {mappedData.length} records
              </p>
            )}
          </CardContent>
        </Card>
      )}

      {/* Supported Formats Guide */}
      <Card className="bg-prop-card border-prop-blue/20">
        <CardHeader>
          <CardTitle className="text-gradient-rainbow">Supported Formats & Auto-Detection</CardTitle>
        </CardHeader>
        <CardContent>
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
            <div className="space-y-3">
              <h4 className="text-prop-blue font-semibold flex items-center">
                <Users className="mr-2 h-4 w-4" />
                Futures Platforms
              </h4>
              <div className="text-sm text-gray-400 space-y-1">
                <p>• Tradovate (auto-detected)</p>
                <p>• Rithmic (auto-detected)</p>
                <p>• CQG (auto-detected)</p>
                <p>• NinjaTrader (auto-detected)</p>
              </div>
            </div>
            <div className="space-y-3">
              <h4 className="text-prop-green font-semibold flex items-center">
                <DollarSign className="mr-2 h-4 w-4" />
                Forex & CFDs
              </h4>
              <div className="text-sm text-gray-400 space-y-1">
                <p>• MT4/MT5 (auto-detected)</p>
                <p>• FTMO (auto-detected)</p>
                <p>• MyFundedFX (auto-detected)</p>
                <p>• TopstepTrader (auto-detected)</p>
              </div>
            </div>
            <div className="space-y-3">
              <h4 className="text-prop-tiffany font-semibold flex items-center">
                <BarChart3 className="mr-2 h-4 w-4" />
                Crypto & Stocks
              </h4>
              <div className="text-sm text-gray-400 space-y-1">
                <p>• Binance (auto-detected)</p>
                <p>• Interactive Brokers (auto-detected)</p>
                <p>• ThinkorSwim (auto-detected)</p>
                <p>• Project X & Custom formats</p>
              </div>
            </div>
          </div>
        </CardContent>
      </Card>
    </div>
  );
}
