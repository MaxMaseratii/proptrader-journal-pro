// 📁 /lib/theme-utils.ts
export const getValueColor = (value: number, type: 'currency' | 'percentage' | 'neutral' = 'currency') => {
  if (type === 'neutral') return colors.neutral.primary;
  if (value > 0) return colors.profit.primary;
  if (value < 0) return colors.loss.primary;
  return colors.neutral.primary;
};

export const getAccountStatusColor = (status: string) => {
  switch (status) {
    case 'active': return colors.account.active;
    case 'funded': return colors.account.funded;
    case 'challenge': return colors.account.challenge;
    case 'failed': return colors.account.failed;
    default: return colors.neutral.primary;
  }
};

export const getRiskColor = (riskPercentage: number) => {
  if (riskPercentage >= 80) return colors.risk.critical;
  if (riskPercentage >= 60) return colors.risk.high;
  if (riskPercentage >= 40) return colors.risk.medium;
  return colors.risk.low;
};

export const getDisciplineGradeColor = (grade: string) => {
  switch (grade) {
    case 'A+':
    case 'A': return colors.profit.primary;
    case 'B': return colors.account.funded;
    case 'C': return colors.account.challenge;
    case 'D': return colors.account.failed;
    case 'F': return colors.risk.critical;
    default: return colors.neutral.primary;
  }
};

// 📁 /components/ui/ProgressBar.tsx
import React from 'react';
import { Progress } from '@/components/ui/progress';
import { cn } from '@/lib/utils';

interface ProgressBarProps {
  value: number;
  max?: number;
  label?: string;
  showPercentage?: boolean;
  color?: 'green' | 'red' | 'blue' | 'orange' | 'purple';
  size?: 'sm' | 'md' | 'lg';
  className?: string;
}

const colorMap = {
  green: 'bg-success-green',
  red: 'bg-error-red',
  blue: 'bg-primary',
  orange: 'bg-warning-orange',
  purple: 'bg-purple-600'
};

const sizeMap = {
  sm: 'h-1.5',
  md: 'h-2',
  lg: 'h-3'
};

export const ProgressBar: React.FC<ProgressBarProps> = ({
  value,
  max = 100,
  label,
  showPercentage = true,
  color = 'blue',
  size = 'md',
  className
}) => {
  const percentage = Math.min((value / max) * 100, 100);
  
  return (
    <div className={cn('w-full', className)}>
      {(label || showPercentage) && (
        <div className="flex justify-between text-sm mb-2">
          {label && <span className="text-gray-300">{label}</span>}
          {showPercentage && (
            <span className={`font-medium ${colorMap[color].replace('bg-', 'text-')}`}>
              {percentage.toFixed(1)}%
            </span>
          )}
        </div>
      )}
      <div className={cn('w-full bg-gray-700 rounded-full', sizeMap[size])}>
        <div 
          className={cn(
            `${sizeMap[size]} rounded-full transition-all duration-300`,
            colorMap[color]
          )}
          style={{ width: `${percentage}%` }}
        />
      </div>
    </div>
  );
};

// 📁 /components/ui/RiskIndicator.tsx
import React from 'react';
import { AlertTriangle, Shield, CheckCircle } from 'lucide-react';
import { cn } from '@/lib/utils';

interface RiskIndicatorProps {
  riskPercentage: number;
  size?: 'sm' | 'md' | 'lg';
  showIcon?: boolean;
  className?: string;
}

export const RiskIndicator: React.FC<RiskIndicatorProps> = ({
  riskPercentage,
  size = 'md',
  showIcon = true,
  className
}) => {
  const getRiskLevel = () => {
    if (riskPercentage >= 80) return 'critical';
    if (riskPercentage >= 60) return 'high';
    if (riskPercentage >= 40) return 'medium';
    return 'low';
  };

  const riskLevel = getRiskLevel();
  
  const config = {
    critical: {
      color: 'text-red-600',
      bgColor: 'bg-red-600',
      icon: AlertTriangle,
      label: 'Critical'
    },
    high: {
      color: 'text-error-red',
      bgColor: 'bg-error-red',
      icon: AlertTriangle,
      label: 'High'
    },
    medium: {
      color: 'text-warning-orange',
      bgColor: 'bg-warning-orange',
      icon: Shield,
      label: 'Medium'
    },
    low: {
      color: 'text-success-green',
      bgColor: 'bg-success-green',
      icon: CheckCircle,
      label: 'Low'
    }
  };

  const { color, bgColor, icon: Icon, label } = config[riskLevel];
  
  return (
    <div className={cn('flex items-center space-x-2', className)}>
      {showIcon && (
        <div className={cn(
          'rounded-full p-1',
          bgColor,
          'bg-opacity-20'
        )}>
          <Icon className={cn(
            'h-4 w-4',
            color,
            size === 'sm' && 'h-3 w-3',
            size === 'lg' && 'h-5 w-5'
          )} />
        </div>
      )}
      <div className={cn(
        'text-sm font-medium',
        color,
        size === 'sm' && 'text-xs',
        size === 'lg' && 'text-base'
      )}>
        {riskPercentage.toFixed(1)}% ({label})
      </div>
    </div>
  );
};
