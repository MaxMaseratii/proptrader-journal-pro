const calculateProjection = async (settings: any) => {
  await new Promise(resolve => setTimeout(resolve, 800));
  
  // Mock calculation results
  const totalDays = Math.floor(settings.profitTarget / (settings.riskPerTrade * settings.riskRewardRatio * 0.6));
  const totalPnl = settings.profitTarget * 1.2;
  
  return {
    summary: {
      totalPnl,
      totalDays,
      targetReached: true,
      winRate: 65,
      totalTrades: totalDays * 2,
    },
    chartData: Array.from({ length: totalDays }, (_, i) => ({
      day: i + 1,
      balance: settings.startingCapital + (i * (totalPnl / totalDays)),
    })),
  };
};

              <CardHeader>
                <div className="flex items-center gap-3 mb-4">
                  <BarChart3 className="w-6 h-6 text-yellow-500" />
                  <div>
                    <CardTitle className="text-white text-xl">Daily Trading Projection</CardTitle>
                    <p className="text-gray-400 text-sm mt-1">
                      Day-by-day breakdown of your path to ${(projectionForm.targetGoalAmount || 0).toLocaleString()} target
                    </p>
                  </div>
                </div>
                
                {/* Key Information Banner */}
                <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-6">
                  <div className="bg-blue-500/10 border border-blue-500/30 rounded-lg p-4">
                    <div className="flex items-center gap-2 mb-2">
                      <Shield className="w-4 h-4 text-blue-400" />
                      <span className="text-blue-400 font-medium text-sm">RISK PER TRADE</span>
                    </div>
                    <div className="text-2xl font-bold text-white">${(projectionSettings.riskPerTrade || 0).toLocaleString()}</div>
                    <div className="text-xs text-gray-400">Amount you risk per trade</div>
                  </div>
                  
                  <div className="bg-green-500/10 border border-green-500/30 rounded-lg p-4">
                    <div className="flex items-center gap-2 mb-2">
                      <Target className="w-4 h-4 text-green-400" />
                      <span className="text-green-400 font-medium text-sm">DAILY PROFIT TARGET</span>
                    </div>
                    <div className="text-2xl font-bold text-white">
                      ${((projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 0) * (1 + (projectionForm.copiedAccounts || 0))).toLocaleString()}
                    </div>
                    <div className="text-xs text-gray-400">
                      {projectionForm.copiedAccounts > 0 ? `Includes ${projectionForm.copiedAccounts} copied accounts` : 'Single account only'}
                    </div>
                  </div>
                  
                  <div className="bg-purple-500/10 border border-purple-500/30 rounded-lg p-4">
                    <div className="flex items-center gap-2 mb-2">
                      <TrendingUp className="w-4 h-4 text-purple-400" />
                      <span className="text-purple-400 font-medium text-sm">ESTIMATED TIMELINE</span>
                    </div>
                    <div className="text-2xl font-bold text-white">
                      {Math.ceil((projectionForm.targetGoalAmount || 0) / (((projectionSettings.riskPerTrade || 1) * (projectionSettings.riskRewardRatio || 1)) * (1 + (projectionForm.copiedAccounts || 0))))} days
                    </div>
                    <div className="text-xs text-gray-400">To reach your target goal</div>
                  </div>
                </div>
              </CardHeader>
              
              <CardContent>
                {/* Calculation Explanation */}
                <div className="bg-gray-800/50 rounded-lg p-4 mb-6">
                  <h4 className="text-white font-semibold mb-2 flex items-center gap-2">
                    <Calculator className="w-4 h-4" />
                    How This Works
                  </h4>
                  <div className="text-sm text-gray-300 space-y-1">
                    <p>• <span className="text-blue-400">Risk per trade:</span> ${(projectionSettings.riskPerTrade || 0).toLocaleString()} × <span className="text-purple-400">{(projectionSettings.riskRewardRatio || 0)}:1 ratio</span> = ${((projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 0)).toLocaleString()} profit per winning trade</p>
                    {projectionForm.copiedAccounts > 0 && (
                      <p>• <span className="text-green-400">With {projectionForm.copiedAccounts} copied accounts:</span> ${((projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 0)).toLocaleString()} × {1 + projectionForm.copiedAccounts} accounts = ${((projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 0) * (1 + projectionForm.copiedAccounts)).toLocaleString()} daily target</p>
                    )}
                    <p>• <span className="text-yellow-400">Target:</span> ${(projectionForm.targetGoalAmount || 0).toLocaleString()} ÷ ${((projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 0) * (1 + projectionForm.copiedAccounts)).toLocaleString()} = {Math.ceil((projectionForm.targetGoalAmount || 0) / (((projectionSettings.riskPerTrade || 1) * (projectionSettings.riskRewardRatio || 1)) * (1 + (projectionForm.copiedAccounts || 0))))} trading days needed</p>
                  </div>
                </div>

                {/* Timeline Table */}
                <div className="bg-gray-800/30 rounded-lg overflow-hidden">
                  <div className="overflow-x-auto">
                    <table className="w-full">
                      <thead className="bg-gray-800/50">
                        <tr>
                          <th className="text-left p-4 text-gray-300 font-semibold">Trading Day</th>
                          <th className="text-center p-4 text-gray-300 font-semibold">
                            <div className="flex flex-col items-center">
                              <span>Risk Amount</span>
                              <span className="text-xs font-normal text-gray-400">(per trade)</span>
                            </div>
                          </th>
                          <th className="text-center p-4 text-gray-300 font-semibold">
                            <div className="flex flex-col items-center">
                              <span>Expected Profit</span>
                              <span className="text-xs font-normal text-gray-400">(if win)</span>
                            </div>
                          </th>
                          <th className="text-center p-4 text-gray-300 font-semibold">
                            <div className="flex flex-col items-center">
                              <span>Progress to Goal</span>
                              <span className="text-xs font-normal text-gray-400">(cumulative)</span>
                            </div>
                          </th>
                          <th className="text-center p-4 text-gray-300 font-semibold">
                            <div className="flex flex-col items-center">
                              <span>Actual Results</span>
                              <span className="text-xs font-normal text-gray-400">(track here)</span>
                            </div>
                          </th>
                        </tr>
                      </thead>
                      <tbody>
                        {Array.from({ length: Math.ceil((projectionForm.targetGoalAmount || 0) / (((projectionSettings.riskPerTrade || 1) * (projectionSettings.riskRewardRatio || 1)) * (1 + (projectionForm.copiedAccounts || 0)))) }, (_, i) => {
                          const day = i + 1;
                          const date = new Date();
                          date.setDate(date.getDate() + i);
                          const totalAccounts = 1 + (projectionForm.copiedAccounts || 0);
                          const dailyProfit = (projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 0) * totalAccounts;
                          const cumulativeProgress = dailyProfit * day;
                          const isTargetReached = cumulativeProgress >= (projectionForm.targetGoalAmount || 0);
                          const progressPercentage = Math.min(100, (cumulativeProgress / (projectionForm.targetGoalAmount || 1)) * 100);
                          
                          return (
                            <tr key={day} className="border-b border-gray-700/50 hover:bg-gray-700/30 transition-colors">
                              <td className="p-4">
                                <div className="flex items-center gap-3">
                                  <div className="flex items-center justify-center w-8 h-8 rounded-full bg-blue-500/20 text-blue-400 text-sm font-semibold">
                                    {day}
                                  </div>
                                  <div>
                                    <div className="text-white font-medium">
                                      {date.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' })}
                                    </div>
                                    <div className="text-xs text-gray-400">Day {day}</div>
                                  </div>
                                </div>
                              </td>
                              <td className="p-4 text-center">
                                <div className="text-blue-400 font-bold text-lg">
                                  ${(projectionSettings.riskPerTrade || 0).toLocaleString()}
                                </div>
                              </td>
                              <td className="p-4 text-center">
                                <div className="text-green-400 font-bold text-lg">
                                  ${dailyProfit.toLocaleString()}
                                </div>
                                {projectionForm.copiedAccounts > 0 && (
                                  <div className="text-xs text-gray-400">
                                    ${((projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 0)).toLocaleString()} × {totalAccounts}
                                  </div>
                                )}
                              </td>
                              <td className="p-4 text-center">
                                <div className="space-y-2">
                                  <div className={`font-bold text-lg ${isTargetReached ? 'text-green-400' : 'text-yellow-400'}`}>
                                    ${Math.min(cumulativeProgress, projectionForm.targetGoalAmount || 0).toLocaleString()}
                                  </div>
                                  <div className="w-full bg-gray-700 rounded-full h-2">
                                    <div 
                                      className={`h-2 rounded-full transition-all ${isTargetReached ? 'bg-green-400' : 'bg-yellow-400'}`}
                                      style={{ width: `${Math.min(100, progressPercentage)}%` }}
                                    ></div>
                                  </div>
                                  <div className="text-xs text-gray-400">
                                    {progressPercentage.toFixed(1)}% of goal
                                  </div>
                                  {isTargetReached && (
                                    <div className="flex items-center justify-center gap-1">
                                      <CheckCircle className="w-4 h-4 text-green-400" />
                                      <span className="text-xs text-green-400 font-medium">TARGET REACHED!</span>
                                    </div>
                                  )}
                                </div>
                              </td>
                              <td className="p-4 text-center">
                                <div className="text-gray-500 font-medium">—</div>
                                <div className="text-xs text-gray-600">Add your results</div>
                              </td>
                            </tr>
                          );
                        })}
                      </tbody>
                    </table>
                  </div>
                </div>
              </CardContent>
            </Card>
          )}

          {!selectedAccount && (
            <Card className="bg-gradient-to-br from-black/95 via-gray-900/90 to-black/95 border border-gray-700">
              <CardContent className="p-12 text-center">
                <Calculator className="w-16 h-16 mx-auto mb-4 text-gray-500" />
                <h3 className="text-xl font-semibold text-white mb-2">Select an Account</h3>
                <p className="text-gray-400 mb-6">Choose an account above to see the projection timeline</p>
              </CardContent>
            </Card>
          )}
      </div>
      
      {/* Enhanced Projection Calculator Dialog */}
      <Dialog open={showProjectionForm} onOpenChange={setShowProjectionForm}>
        <DialogContent className="max-w-7xl max-h-[95vh] overflow-y-auto bg-gray-900 border-gray-700">
          <DialogHeader>
            <DialogTitle className="text-white text-xl">Projection Calculator</DialogTitle>
          </DialogHeader>
          
          <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
            {/* Settings Column */}
            <div className="space-y-4">
              <h3 className="text-lg font-semibold text-white">Projection Settings</h3>
              
              <div>
                <Label className="text-white">Starting Capital (Display Only)</Label>
                <div className="bg-gray-800 border border-gray-600 rounded-md p-3 text-white text-lg font-semibold">
                  {(projectionSettings.startingCapital || 0).toLocaleString('en-US', {
                    style: 'currency',
                    currency: 'USD',
                    minimumFractionDigits: 0,
                    maximumFractionDigits: 0
                  })}
                </div>
              </div>
              
                <div>
                  <Label className="text-white">Copied Account Number</Label>
                  <Input 
                    type="number"
                    min="0"
                    value={projectionForm.copiedAccounts}
                    onChange={(e) => setProjectionForm({...projectionForm, copiedAccounts: parseInt(e.target.value) || 0})}
                    className="bg-gray-800 border-gray-600 text-white"
                    placeholder="# of similar accounts"
                  />
                  <p className="text-xs text-gray-400 mt-1">Connect multiple similar accounts for combined projection</p>
                </div>
                
                <div>
                  <Label className="text-white">Target Goal Amount ($)</Label>
                  <Input 
                    type="number"
                    value={projectionForm.targetGoalAmount}
                    onChange={(e) => setProjectionForm({...projectionForm, targetGoalAmount: parseFloat(e.target.value) || 0})}
                    className="bg-gray-800 border-gray-600 text-white"
                  />
                </div>
              
              <div>
                <Label className="text-white">Risk Per Trade</Label>
                <Input 
                  type="number"
                  value={projectionSettings.riskPerTrade || 0}
                  onChange={(e) => setProjectionSettings(prev => ({
                    ...prev,
                    riskPerTrade: Number(e.target.value)
                  }))}
                  className="bg-gray-800 border-gray-600 text-white"
                />
              </div>
              
              <div>
                <Label className="text-white">Risk:Reward Ratio</Label>
                <Input 
                  type="number"
                  step="0.1"
                  value={projectionSettings.riskRewardRatio || 0}
                  onChange={(e) => setProjectionSettings(prev => ({
                    ...prev,
                    riskRewardRatio: Number(e.target.value)
                  }))}
                  className="bg-gray-800 border-gray-600 text-white"
                />
              </div>
              
                <div>
                 <Label className="text-white">Total Profit (Auto-calculated)</Label>
                 <div className="bg-gray-800 border border-gray-600 rounded-md p-3 text-white text-lg font-semibold">
                   {calculateTotalProfit(
                     projectionSettings.riskPerTrade || 0,
                     projectionSettings.riskRewardRatio || 1,
                     projectionForm.copiedAccounts || 0
                   ).toLocaleString('en-US', {
                     style: 'currency',
                     currency: 'USD',
                     minimumFractionDigits: 0,
                     maximumFractionDigits: 0
                   })}
                 </div>
                 <p className="text-xs text-gray-400 mt-1">Per trade profit across all accounts</p>
               </div>
              
              <div>
                <Label className="text-white">Max Drawdown</Label>
                <Input 
                  type="number"
                  value={projectionSettings.maxDrawdown || 0}
                  onChange={(e) => setProjectionSettings(prev => ({
                    ...prev,
                    maxDrawdown: Number(e.target.value)
                  }))}
                  className="bg-gray-800 border-gray-600 text-white"
                />
              </div>
              
              <div>
                <Label className="text-white">Max Loss/Day</Label>
                <Input 
                  type="number"
                  value={projectionSettings.maxLossPerDay || 0}
                  onChange={(e) => setProjectionSettings(prev => ({
                    ...prev,
                    maxLossPerDay: Number(e.target.value)
                  }))}
                  className="bg-gray-800 border-gray-600 text-white"
                />
              </div>
              
              <div>
                <Label className="text-white flex items-center gap-2">
                  Risk Cutting 🛡️ - {projectionSettings.riskCuttingPercent || 0}%
                </Label>
                <Slider
                  value={[projectionSettings.riskCuttingPercent || 0]}
                  onValueChange={([value]) => setProjectionSettings(prev => ({
                    ...prev,
                    riskCuttingPercent: value
                  }))}
                  max={50}
                  step={5}
                  className="w-full"
                />
                <p className="text-xs text-gray-400">Reduce risk after consecutive losses</p>
              </div>
              
              <div>
                <Label className="text-white flex items-center gap-2">
                  Compounding 📊 - {projectionSettings.compoundingPercent || 0}%
                </Label>
                <Slider
                  value={[projectionSettings.compoundingPercent || 0]}
                  onValueChange={([value]) => setProjectionSettings(prev => ({
                    ...prev,
                    compoundingPercent: value
                  }))}
                  max={30}
                  step={5}
                  className="w-full"
                />
                <p className="text-xs text-gray-400">Increase risk when profitable</p>
              </div>
            </div>
            
            {/* Live Preview Column */}
            <div className="space-y-4">
              <h3 className="text-lg font-semibold text-white">Live Effect Preview</h3>
              
              {/* Real-time Impact Cards */}
              <div className="space-y-3">
                 <div className="bg-gradient-to-r from-blue-900/20 to-blue-800/20 border border-blue-500/30 p-4 rounded-lg">
                   <div className="text-sm text-gray-400">Daily Target</div>
                   <div className="text-xl font-bold text-blue-400">
                     {calculateTotalProfit(
                       projectionSettings.riskPerTrade || 0,
                       projectionSettings.riskRewardRatio || 1,
                       projectionForm.copiedAccounts || 0
                     ).toLocaleString('en-US', {
                       style: 'currency',
                       currency: 'USD',
                       minimumFractionDigits: 0
                     })}
                   </div>
                   {(projectionForm.copiedAccounts || 0) > 0 && (
                     <div className="text-xs text-green-400">
                       ×{1 + (projectionForm.copiedAccounts || 0)} accounts total
                     </div>
                   )}
                 </div>
                
                <div className="bg-gradient-to-r from-green-900/20 to-green-800/20 border border-green-500/30 p-4 rounded-lg">
                  <div className="text-sm text-gray-400">Adjusted Risk</div>
                  <div className="text-xl font-bold text-green-400">
                    {((projectionSettings.riskPerTrade || 0) * (1 - (projectionSettings.riskCuttingPercent || 0) / 100)).toLocaleString('en-US', {
                      style: 'currency',
                      currency: 'USD',
                      minimumFractionDigits: 0
                    })}
                  </div>
                  <div className="text-xs text-gray-400">
                    {(projectionSettings.riskCuttingPercent || 0) > 0 ? `${projectionSettings.riskCuttingPercent}% reduction applied` : 'No risk cutting'}
                  </div>
                </div>
                
                 <div className="bg-gradient-to-r from-purple-900/20 to-purple-800/20 border border-purple-500/30 p-4 rounded-lg">
                   <div className="text-sm text-gray-400">Compounded Daily Target</div>
                   <div className="text-xl font-bold text-purple-400">
                     {(calculateTotalProfit(
                       projectionSettings.riskPerTrade || 0,
                       projectionSettings.riskRewardRatio || 1,
                       projectionForm.copiedAccounts || 0
                     ) * (1 + (projectionForm.compoundingPercent || 0) / 100)).toLocaleString('en-US', {
                       style: 'currency',
                       currency: 'USD',
                       minimumFractionDigits: 0
                     })}
                   </div>
                   <div className="text-xs text-gray-400">
                     {(projectionForm.compoundingPercent || 0) > 0 ? `${projectionForm.compoundingPercent}% compounding boost` : 'No compounding'}
                   </div>
                 </div>
                
                 <div className="bg-gradient-to-r from-yellow-900/20 to-yellow-800/20 border border-yellow-500/30 p-4 rounded-lg">
                   <div className="text-sm text-gray-400">Days to Target</div>
                   <div className="text-xl font-bold text-yellow-400">
                      {Math.ceil((projectionForm.targetGoalAmount || 0) / Math.max(1, calculateTotalProfit(
                        projectionSettings.riskPerTrade || 0,
                        projectionSettings.riskRewardRatio || 1,
                        projectionForm.copiedAccounts || 0
                      ) * (1 + (projectionForm.compoundingPercent || 0) / 100)))}
                   </div>
                   <div className="text-xs text-gray-400">
                     Estimated timeline
                   </div>
                 </div>
              </div>
              
              {/* Risk Level Indicator */}
              <div className="bg-gray-800 p-4 rounded-lg">
                <div className="text-sm text-gray-400 mb-2">Risk Level</div>
                <div className="flex items-center gap-2">
                  {(() => {
                    const riskPercentage = ((projectionSettings.riskPerTrade || 0) / (projectionSettings.startingCapital || 1)) * 100;
                    if (riskPercentage <= 1) return <span className="text-green-400">🛡️ Conservative</span>;
                    if (riskPercentage <= 2) return <span className="text-yellow-400">📈 Moderate</span>;
                    if (riskPercentage <= 3) return <span className="text-orange-400">🚀 Aggressive</span>;
                    return <span className="text-red-400">🔥 Extreme</span>;
                  })()}
                </div>
                <div className="w-full bg-gray-700 rounded-full h-2 mt-2">
                  <div 
                    className={`h-2 rounded-full transition-all duration-300 ${
                      ((projectionSettings.riskPerTrade || 0) / (projectionSettings.startingCapital || 1)) * 100 <= 1 ? 'bg-green-400' :
                      ((projectionSettings.riskPerTrade || 0) / (projectionSettings.startingCapital || 1)) * 100 <= 2 ? 'bg-yellow-400' :
                      ((projectionSettings.riskPerTrade || 0) / (projectionSettings.startingCapital || 1)) * 100 <= 3 ? 'bg-orange-400' : 'bg-red-400'
                    }`}
                    style={{ 
                      width: `${Math.min(100, ((projectionSettings.riskPerTrade || 0) / (projectionSettings.startingCapital || 1)) * 100 * 25)}%` 
                    }}
                  />
                </div>
              </div>
            </div>
            
            {/* Results Column */}
            <div className="space-y-4">
              <h3 className="text-lg font-semibold text-white">Projection Results</h3>
              
              <div className="bg-gray-800 p-4 rounded-lg">
                <div className="text-3xl font-bold text-green-400">
                  {((projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 1) * (projectionSettings.accountMultiplier || 1) * Math.ceil((projectionForm.targetGoalAmount || 0) / Math.max(1, (projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 1) * (projectionSettings.accountMultiplier || 1)))).toLocaleString('en-US', {
                    style: 'currency',
                    currency: 'USD',
                    minimumFractionDigits: 0,
                    maximumFractionDigits: 0
                  })}
                </div>
                <div className="text-gray-400">Total P&L</div>
              </div>
              
              <div className="bg-gray-800 p-4 rounded-lg">
                <div className="text-3xl font-bold text-blue-400">
                  {Math.ceil((projectionForm.targetGoalAmount || 0) / Math.max(1, (projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 1) * (projectionSettings.accountMultiplier || 1)))}
                </div>
                <div className="text-gray-400">Days to Target</div>
              </div>
              
              <div className="text-green-400 text-sm bg-green-900/20 p-3 rounded-lg border border-green-500/30">
                ✅ Target achievable with these settings
              </div>
              
              {/* Account Multiplier Info */}
              {(projectionSettings.accountMultiplier || 1) > 1 && (
                <div className="bg-blue-900/20 p-3 rounded-lg border border-blue-500/30">
                  <div className="text-blue-400 font-semibold">Multi-Account Projection</div>
                  <div className="text-sm text-gray-300">
                    {projectionSettings.accountMultiplier} accounts (1 original + {(projectionSettings.accountMultiplier || 1) - 1} copied)
                  </div>
                  <div className="text-xs text-gray-400 mt-1">
                    Combined daily target: {((projectionSettings.riskPerTrade || 0) * (projectionSettings.riskRewardRatio || 1) * (projectionSettings.accountMultiplier || 1)).toLocaleString('en-US', {
                      style: 'currency',
                      currency: 'USD',
                      minimumFractionDigits: 0
                    })}
                  </div>
                </div>
              )}
              
              {/* Validation Warnings */}
              <div className="space-y-2">
                {((projectionSettings.riskPerTrade || 0) > (projectionSettings.maxLossPerDay || Infinity)) && (
                  <div className="bg-red-900/20 p-3 rounded-lg border border-red-500/30">
                    <div className="text-red-400 text-sm">⚠️ Risk per trade exceeds daily loss limit</div>
                  </div>
                )}
                
                {((projectionSettings.riskPerTrade || 0) / (projectionSettings.startingCapital || 1)) * 100 > 5 && (
                  <div className="bg-orange-900/20 p-3 rounded-lg border border-orange-500/30">
                    <div className="text-orange-400 text-sm">⚠️ High risk percentage detected</div>
                  </div>
                )}
              </div>
            </div>
          </div>
          
          <div className="flex justify-end gap-2 pt-4">
            <Button variant="outline" onClick={() => setShowProjectionForm(false)}>
              Cancel
            </Button>
            <Button 
              onClick={() => selectedAccount && handleSaveProjection(selectedAccount)}
              disabled={!selectedAccount || createProjectionMutation.isPending}
              className="bg-yellow-500 hover:bg-yellow-600 text-black"
            >
              {createProjectionMutation.isPending ? 'Saving...' : 'Save Projection'}
            </Button>
          </div>
        </DialogContent>
      </Dialog>
    </div>
  );
};

export default Projections;
