get_period_risk(capital, period, custom_risk) =>
    period_days = switch period
        'Weekly (5 days)' => 5.0
        'BiWeekly (10 days)' => 10.0
        'Weekdays (20 days)' => 20.0
        'Calendar Month (30 days)' => 30.0
        'Custom Risk Amount' => 1.0
        => 10.0
    
    period == 'Custom Risk Amount' ? custom_risk : capital / period_days

calculate_risk_score(capital, total_risk, risk_pct) =>
    base_score = math.min(risk_pct * 10, 50)
    capital_factor = capital < 5000 ? 20 : (capital < 25000 ? 10 : 0)
    risk_ratio = total_risk / capital * 100
    risk_factor = risk_ratio > 5 ? 30 : (risk_ratio > 2 ? 15 : 0)
    score = base_score + capital_factor + risk_factor
    math.min(math.round(score), 100)

get_risk_color(risk_score) =>
    if not use_dynamic_colors
        primary_row_color
    else
        if risk_score <= 30
            success_color_custom
        else if risk_score <= 60
            warning_color_custom
        else
            danger_color_custom

get_text_size(size_str) =>
    switch size_str
        'Tiny' => size.tiny

        'Normal' => size.normal
        'Large' => size.large
        'Huge' => size.huge
        => size.normal

calculate_max_size(margin_req, capital) =>
    if margin_req <= 0
        0
    else 
        available_capital = capital * (1 - margin_buffer_pct / 100)
        math.floor(available_capital / margin_req)

calculate_safe_size(lots, capital, point_sl, point_value, use_risk_pct, risk_pct) =>
    if point_value <= 0
        0
    else 
        risk_per_contract = point_sl * point_value
        max_risk = use_risk_pct ? capital * (risk_pct / 100) : capital * 0.02
        positions_by_risk = math.floor(max_risk / risk_per_contract)
        math.min(lots, positions_by_risk)

calculate_position_risk(contracts, point_sl, point_value) =>
    contracts * point_sl * point_value

format_currency(value) =>
    '$' + str.tostring(value, '#,##0.' + str.repeat('#', decimal_places))
